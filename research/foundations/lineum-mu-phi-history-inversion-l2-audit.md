from __future__ import annotations

import itertools

import numpy as np


@dataclass(frozen=True)
class Config:
    dt: float = 1.0
    psi_diffusion: float = 0.05
    phi_diffusion: float = 0.05
    drift_strength: float = -0.004
    stencil_type: str = "LAP4"
    phi_diffusion_scales_with_dt: bool = False
    use_mode_coupling: bool = True
    mode_coupling_strength: float = 0.001
    mu_eta: float = 0.005
    mu_rho: float = 0.0001
    mu_cap: float = 10.0
    mu_peak_cutoff_ratio: float = 0.1
    psi_amp_cap: float = 1e6
    grad_cap: float = 1e6
    phi_cap: float = 1e6


def diffuse(field, kappa, rate, stencil_type):
    k_up = np.roll(kappa, 1, axis=0)
    k_dn = np.roll(kappa, -1, axis=0)
    k_lf = np.roll(kappa, 1, axis=1)
    k_rt = np.roll(kappa, -1, axis=1)

    f_up = np.roll(field, 1, axis=0)
    f_dn = np.roll(field, -1, axis=0)
    f_lf = np.roll(field, 1, axis=1)
    f_rt = np.roll(field, -1, axis=1)

    if stencil_type == "LAP8":
        k_ul = np.roll(k_up, 1, axis=1)
        k_ur = np.roll(k_up, -1, axis=1)
        k_dl = np.roll(k_dn, 1, axis=1)
        k_dr = np.roll(k_dn, -1, axis=1)

        f_ul = np.roll(f_up, 1, axis=1)
        f_ur = np.roll(f_up, -1, axis=1)
        f_dl = np.roll(f_dn, 1, axis=1)
        f_dr = np.roll(f_dn, -1, axis=1)

        neighbor_sum = (
            f_up * k_up
            + f_dn * k_dn
            + f_lf * k_lf
            + f_rt * k_rt
            + 0.25
            * (
                f_ul * k_ul
                + f_ur * k_ur
                + f_dl * k_dl
                + f_dr * k_dr
            )
        )
        active_neighbors = (
            k_up
            + k_dn
            + k_lf
            + k_rt
            + 0.25 * (k_ul + k_ur + k_dl + k_dr)
        )
    else:
        neighbor_sum = (
            f_up * k_up + f_dn * k_dn + f_lf * k_lf + f_rt * k_rt
        )
        active_neighbors = k_up + k_dn + k_lf + k_rt

    return rate * (neighbor_sum - active_neighbors * field)


def cap_complex_magnitude(z, cap):
    z = np.asarray(z, dtype=np.complex128).copy()
    magnitude = np.abs(z)
    mask = magnitude > cap
    if np.any(mask):
        z[mask] *= cap / (magnitude[mask] + 1e-30)
    return z


def deterministic_step(state, cfg, write_mu=True):
    psi = np.asarray(state["psi"], dtype=np.complex128).copy()
    phi = np.asarray(state["phi"], dtype=np.float64).copy()
    kappa = np.asarray(state["kappa"], dtype=np.float64).copy()
    mu = np.asarray(state["mu"], dtype=np.float64).copy()

    amplitude = np.clip(np.abs(psi), 0.0, cfg.psi_amp_cap)
    grad_x, grad_y = np.gradient(amplitude)
    grad_x = np.clip(grad_x, -cfg.grad_cap, cfg.grad_cap)
    grad_y = np.clip(grad_y, -cfg.grad_cap, cfg.grad_cap)

    drift_multiplier = 1.0 + mu

    phi_interaction = np.clip(phi, 0.0, 10.0)
    interaction_factor = 0.1 * np.tanh(
        (0.04 * phi_interaction * kappa * drift_multiplier) / 0.1
    )
    interaction_term = interaction_factor * psi
    interaction_magnitude = np.abs(interaction_term)
    interaction_term /= 1.0 + interaction_magnitude / 10.0

    grad_phi_x, grad_phi_y = np.gradient(phi)
    phi_flow_term = (
        cfg.drift_strength
        * (grad_phi_x + 1j * grad_phi_y)
        * kappa
        * drift_multiplier
    )
    flow_magnitude = np.abs(phi_flow_term)
    phi_flow_term /= 1.0 + flow_magnitude / 10.0

    psi += phi_flow_term * cfg.dt
    psi = cap_complex_magnitude(psi, cfg.psi_amp_cap)
    psi += interaction_term * cfg.dt
    psi -= 0.005 * psi * cfg.dt
    psi += (
        diffuse(
            psi,
            kappa,
            rate=cfg.psi_diffusion,
            stencil_type=cfg.stencil_type,
        )
        * kappa
        * cfg.dt
    )

    write_energy = np.abs(psi) ** 2

    if not cfg.use_mode_coupling:
        raise ValueError("This audit requires mode coupling.")

    delta_energy = (
        cfg.mode_coupling_strength
        * write_energy
        * kappa
        * cfg.dt
    )
    phi += delta_energy

    psi_new_magnitude = np.sqrt(
        np.maximum(write_energy - delta_energy, 0.0)
    )
    psi = (
        psi / (np.sqrt(write_energy) + 1e-12)
    ) * psi_new_magnitude

    phi_step_scale = (
        cfg.dt if cfg.phi_diffusion_scales_with_dt else 1.0
    )
    phi += (
        kappa
        * cfg.phi_diffusion
        * diffuse(
            phi,
            kappa,
            rate=0.05,
            stencil_type=cfg.stencil_type,
        )
        * phi_step_scale
    )
    phi = np.clip(phi, 0.0, cfg.phi_cap)

    if write_mu:
        mu = update_mu_from_energy(mu, write_energy, kappa, cfg)

    return {
        "psi": psi,
        "phi": phi,
        "kappa": kappa,
        "mu": mu,
        "write_energy": write_energy,
    }


def update_mu_from_energy(mu, energy, kappa, cfg):
    mu = np.asarray(mu, dtype=np.float64).copy()
    energy = np.asarray(energy, dtype=np.float64)

    dynamic_floor = cfg.mu_peak_cutoff_ratio
    if 0.0 < dynamic_floor < 1.0:
        dynamic_floor *= np.max(energy)

    activity = np.maximum(energy - dynamic_floor, 0.0)
    drift_multiplier = 1.0 + mu
    mu += (
        cfg.mu_eta
        * activity
        * kappa
        * drift_multiplier
        * cfg.dt
    )
    mu -= cfg.mu_rho * mu * cfg.dt
    return np.clip(mu, 0.0, cfg.mu_cap)


def laplacian_symbol(size, stencil_type):
    frequencies = 2.0 * np.pi * np.fft.fftfreq(size)
    ky, kx = np.meshgrid(frequencies, frequencies, indexing="ij")

    if stencil_type == "LAP8":
        return (
            2.0 * np.cos(kx)
            + 2.0 * np.cos(ky)
            + np.cos(kx) * np.cos(ky)
            - 5.0
        )

    return 2.0 * np.cos(kx) + 2.0 * np.cos(ky) - 4.0


def phi_transfer_multiplier(size, cfg):
    step_scale = (
        cfg.dt if cfg.phi_diffusion_scales_with_dt else 1.0
    )
    coefficient = cfg.phi_diffusion * 0.05 * step_scale
    return 1.0 + coefficient * laplacian_symbol(
        size,
        cfg.stencil_type,
    )


def recover_energy_from_phi(phi_before, phi_after, cfg):
    multiplier = phi_transfer_multiplier(phi_before.shape[0], cfg)
    if np.min(np.abs(multiplier)) <= 1e-12:
        raise ValueError("Phi transport is singular or unsafe to invert.")

    phi_mid = np.fft.ifft2(
        np.fft.fft2(phi_after) / multiplier
    ).real

    energy = (
        (phi_mid - phi_before)
        / (cfg.mode_coupling_strength * cfg.dt)
    )
    energy = np.maximum(energy, 0.0)
    return energy, phi_mid, multiplier


def make_initial(kind, size=32, seed=314159):
    y, x = np.mgrid[0:size, 0:size]
    center = (size - 1) / 2.0

    if kind == "gaussian":
        radius_squared = (x - center) ** 2 + (y - center) ** 2
        amplitude = 0.7 * np.exp(-radius_squared / (2.0 * 4.5**2))
        return amplitude.astype(np.complex128)

    if kind == "phase_winding":
        radius_squared = (x - center) ** 2 + (y - center) ** 2
        amplitude = 0.7 * np.exp(-radius_squared / (2.0 * 5.0**2))
        phase = np.arctan2(y - center, x - center)
        return amplitude * np.exp(1j * phase)

    if kind == "two_packets":
        radius_a = (x - 10.0) ** 2 + (y - center) ** 2
        radius_b = (x - 22.0) ** 2 + (y - center) ** 2
        packet_a = 0.5 * np.exp(-radius_a / (2.0 * 3.0**2))
        packet_b = 0.45 * np.exp(-radius_b / (2.0 * 3.5**2))
        return packet_a + packet_b * np.exp(0.8j)

    if kind == "smooth_random":
        rng = np.random.default_rng(seed)
        raw = rng.normal(size=(size, size)) + 1j * rng.normal(
            size=(size, size)
        )
        spectrum = np.fft.fft2(raw)
        frequencies = np.fft.fftfreq(size)
        ky, kx = np.meshgrid(frequencies, frequencies, indexing="ij")
        low_pass = np.exp(-((kx**2 + ky**2) / 0.02))
        field = np.fft.ifft2(spectrum * low_pass)
        field *= 0.35 / (np.max(np.abs(field)) + 1e-30)
        return field

    raise ValueError(kind)


def initial_state(kind, size=32):

        "psi": make_initial(kind, size=size),
        "phi": np.zeros((size, size), dtype=np.float64),
        "kappa": np.ones((size, size), dtype=np.float64),

def passive_case(kind, cfg, steps=80):
    state = initial_state(kind)
    replay_mu = np.zeros_like(state["mu"])

    maximum_energy_error = 0.0
    maximum_mu_error = 0.0
    maximum_naive_no_inverse_error = 0.0
    maximum_phi = 0.0

    multiplier = phi_transfer_multiplier(
        state["phi"].shape[0],
        cfg,
    )

    for _ in range(steps):
        phi_before = state["phi"].copy()
        next_state = deterministic_step(state, cfg, write_mu=True)

        recovered_energy, _, _ = recover_energy_from_phi(
            phi_before,
            next_state["phi"],
            cfg,
        )
        replay_mu = update_mu_from_energy(
            replay_mu,
            recovered_energy,
            state["kappa"],
            cfg,
        )

        naive_energy = np.maximum(
            (next_state["phi"] - phi_before)
            / (cfg.mode_coupling_strength * cfg.dt),
            0.0,
        )
        naive_mu = update_mu_from_energy(
            state["mu"],
            naive_energy,
            state["kappa"],
            cfg,
        )

        maximum_energy_error = max(
            maximum_energy_error,
            float(
                np.max(
                    np.abs(
                        recovered_energy
                        - next_state["write_energy"]
                    )
                )
            ),
        )
        maximum_mu_error = max(
            maximum_mu_error,
            float(np.max(np.abs(replay_mu - next_state["mu"]))),
        )
        maximum_naive_no_inverse_error = max(
            maximum_naive_no_inverse_error,
            float(np.max(np.abs(naive_mu - next_state["mu"]))),
        )
        maximum_phi = max(
            maximum_phi,
            float(np.max(next_state["phi"])),
        )
        state = next_state

    return {
        "initialization": kind,
        "dt": cfg.dt,
        "stencil": cfg.stencil_type,
        "phi_diffusion_scales_with_dt": (
            cfg.phi_diffusion_scales_with_dt
        ),
        "maximum_energy_error": maximum_energy_error,
        "maximum_mu_error": maximum_mu_error,

            maximum_naive_no_inverse_error

        "maximum_phi": maximum_phi,
        "minimum_transfer_magnitude": float(

        "condition_number": float(
            np.max(np.abs(multiplier))
            / np.min(np.abs(multiplier))

## 27. Final scope statement

The strongest retained result is computational and observer-relative:

```text
when the complete Phi record is lossless and its transport is invertible,
current mu can be reconstructed and replaced exactly;

when the Phi observer loses information,
the missing completion is real relative to that observer,
but its mechanism must be identified rather than named by metaphor.
```

The Horus-eye/Heka story supplied a productive research picture: a finite observer can hold nearly a whole description while a residual completes the causal account. The evidence does not say what that residual is in nature. In current Lineum it already has at least two ordinary implementation causes—clipping and hidden threshold input—before any quantum, cross-domain, or cosmological explanation is needed.
