def make_graph(edges):
    i = np.array([edge[0] for edge in edges], dtype=int)
    j = np.array([edge[1] for edge in edges], dtype=int)
    rest = np.linalg.norm(REFERENCE[j] - REFERENCE[i], axis=1)
    return i, j, rest


FULL_I, FULL_J, FULL_REST = make_graph(FULL_EDGES)
CUT_I, CUT_J, CUT_REST = make_graph(CUT_EDGES)


def wrap_phase(value):
    return (value + np.pi) % (2.0 * np.pi) - np.pi


def grid_coordinates(grid):
    axis = np.linspace(-DOMAIN_HALF, DOMAIN_HALF, grid)
    return axis, np.meshgrid(axis, axis, indexing="xy")


def render_field(points, charges, grid=GRID, global_phase=0.0):
    _, (xx, yy) = grid_coordinates(grid)
    phase = np.full_like(xx, float(global_phase), dtype=float)
    amplitude = np.ones_like(xx, dtype=float)
    for point, charge in zip(points, charges):
        dx = xx - point[0]
        dy = yy - point[1]
        distance = np.hypot(dx, dy)
        phase += charge * np.arctan2(dy, dx)
        amplitude *= np.tanh(distance / CORE_RADIUS)
    return amplitude * np.exp(1j * phase)


def detect_vortices_vectorized(psi):
    phase = np.angle(psi)
    p00 = phase[:-1, :-1]
    p01 = phase[:-1, 1:]
    p11 = phase[1:, 1:]
    p10 = phase[1:, :-1]
    winding = wrap_phase(p01 - p00) + wrap_phase(p11 - p01) + wrap_phase(p10 - p11) + wrap_phase(p00 - p10)
    charge = np.rint(winding / (2.0 * np.pi)).astype(int)
    rows, cols = np.where(charge != 0)
    axis, _ = grid_coordinates(psi.shape[0])
    x = 0.5 * (axis[cols] + axis[cols + 1])
    y = 0.5 * (axis[rows] + axis[rows + 1])
    points = np.column_stack((x, y)) if len(rows) else np.empty((0, 2))
    return points, charge[rows, cols]


def detect_vortices_loop(psi):
    phase = np.angle(psi)
    axis, _ = grid_coordinates(psi.shape[0])
    points = []
    charges = []
    for row in range(psi.shape[0] - 1):
        for col in range(psi.shape[1] - 1):
            corners = [phase[row, col], phase[row, col + 1], phase[row + 1, col + 1], phase[row + 1, col], phase[row, col]]
            winding = sum(wrap_phase(corners[k + 1] - corners[k]) for k in range(4))
            charge = int(np.rint(winding / (2.0 * np.pi)))
            if charge:
                points.append([0.5 * (axis[col] + axis[col + 1]), 0.5 * (axis[row] + axis[row + 1])])
                charges.append(charge)
    return np.asarray(points, dtype=float), np.asarray(charges, dtype=int)


def sorted_detection(points, charges):
    return sorted((int(charge), round(float(point[0]), 12), round(float(point[1]), 12)) for point, charge in zip(points, charges))


def proper_rotation(source, target):
    source_centered = source - source.mean(axis=0)
    target_centered = target - target.mean(axis=0)
    u, _, vt = np.linalg.svd(source_centered.T @ target_centered)
    rotation = u @ vt
    if np.linalg.det(rotation) < 0.0:
        u[:, -1] *= -1.0
        rotation = u @ vt
    return rotation


def invariant_geometry_error(source, source_charges, target, target_charges):
    if len(source) != len(target) or len(source) == 0:
        return math.inf, math.nan
    if sorted(source_charges.tolist()) != sorted(target_charges.tolist()):
        return math.inf, math.nan
    unique_charges = sorted(set(target_charges.tolist()))
    target_groups = [np.where(target_charges == charge)[0] for charge in unique_charges]
    source_groups = [np.where(source_charges == charge)[0] for charge in unique_charges]
    options = [list(itertools.permutations(group.tolist())) for group in source_groups]
    target_scale = np.sqrt(np.mean(np.sum((target - target.mean(axis=0)) ** 2, axis=1)))
    target_scale = max(float(target_scale), 1e-12)
    best_error = math.inf
    best_angle = math.nan
    for choices in itertools.product(*options):
        ordered = np.empty_like(target, dtype=float)
        for target_indices, source_indices in zip(target_groups, choices):
            for target_index, source_index in zip(target_indices, source_indices):
                ordered[target_index] = source[source_index]
        rotation = proper_rotation(ordered, target)
        aligned = (ordered - ordered.mean(axis=0)) @ rotation
        centered_target = target - target.mean(axis=0)
        error = float(np.sqrt(np.mean(np.sum((aligned - centered_target) ** 2, axis=1))) / target_scale)
        if error < best_error:
            best_error = error
            best_angle = math.degrees(math.atan2(rotation[1, 0], rotation[0, 0]))
    return best_error, best_angle


def localization_error(observed, observed_charges, latent, latent_charges):
    if len(observed) != len(latent):
        return math.inf
    maximum = 0.0
    for charge in sorted(set(latent_charges.tolist())):
        observed_indices = np.where(observed_charges == charge)[0]
        latent_indices = np.where(latent_charges == charge)[0]
        if len(observed_indices) != len(latent_indices):
            return math.inf
        distance = np.linalg.norm(observed[observed_indices, None, :] - latent[None, latent_indices, :], axis=2)
        rows, cols = linear_sum_assignment(distance)
        maximum = max(maximum, float(np.max(distance[rows, cols])))
    return maximum


def local_force_vectorized(x, v, edge_i, edge_j, rest):
    delta = x[edge_j] - x[edge_i]
    distance = np.linalg.norm(delta, axis=1)
    unit = np.divide(delta, distance[:, None], out=np.zeros_like(delta), where=distance[:, None] > 1e-15)
    pair_force = K * (distance - rest)[:, None] * unit
    force = np.zeros_like(x)
    np.add.at(force, edge_i, pair_force)
    np.add.at(force, edge_j, -pair_force)
    force += -GAMMA * (v - v.mean(axis=0))
    return force


def local_force_loop(x, v, edges, rest):
    force = np.zeros_like(x)
    for (i, j), rest_length in zip(edges, rest):
        delta = x[j] - x[i]
        distance = float(np.linalg.norm(delta))
        if distance > 1e-15:
            pair_force = K * (distance - rest_length) * delta / distance
            force[i] += pair_force
            force[j] -= pair_force
    force += -GAMMA * (v - v.mean(axis=0))
    return force


def global_force(x, v):
    return -K * ((x - x.mean(axis=0)) - REFERENCE) - GAMMA * (v - v.mean(axis=0))


def initial_state(seed):
    rng = np.random.default_rng(seed)
    transform = np.array([[1.18, 0.10], [0.04, 0.84]], dtype=float)
    x = REFERENCE @ transform.T + rng.normal(0.0, 0.04, REFERENCE.shape)
    x -= x.mean(axis=0)
    return x, np.tile(BULK, (N, 1))


def form_state(seed, dt, mode, force_impl="vectorized"):
    x, v = initial_state(seed)
    for _ in range(int(round(T_FORM / dt))):
        if mode == "local":
            force = local_force_vectorized(x, v, FULL_I, FULL_J, FULL_REST) if force_impl == "vectorized" else local_force_loop(x, v, FULL_EDGES, FULL_REST)
        else:
            force = global_force(x, v)
        v += dt * force
        x += dt * v
    return x


def case_spec(case):
    if case == "L_INTACT": return "local", "local", FULL_I, FULL_J, FULL_REST, FULL_EDGES
    if case == "L_CUT": return "local", "local", CUT_I, CUT_J, CUT_REST, CUT_EDGES
    if case == "L_REMOVED": return "local", "zero", None, None, None, None
    if case in ("G_CONTINUOUS", "G_CUT"): return "global", "global", None, None, None, None
    if case == "G_REMOVED": return "global", "zero", None, None, None, None
    if case == "REGENERATED": return "local", "regen", None, None, None, None
    raise ValueError(case)

def observe(points, grid=GRID, global_phase=0.0, detector="vectorized"):
    psi = render_field(points, CHARGES, grid=grid, global_phase=global_phase)
    observed, charges = detect_vortices_vectorized(psi) if detector == "vectorized" else detect_vortices_loop(psi)
    return {"points": observed, "charges": charges, "localization": localization_error(observed, charges, points, CHARGES), "exact": len(observed) == N and sorted(charges.tolist()) == sorted(CHARGES.tolist())}


def continuity_score(frames):
    if not frames: return 0.0
    previous_points, previous_charges = frames[0]
    alive = np.ones(len(previous_points), dtype=bool)
    for current_points, current_charges in frames[1:]:
        if len(current_points) != len(previous_points) or sorted(current_charges.tolist()) != sorted(previous_charges.tolist()): return 0.0
        current_alive = np.zeros(len(current_points), dtype=bool)
        for charge in sorted(set(previous_charges.tolist())):
            previous_indices = np.where(previous_charges == charge)[0]
            current_indices = np.where(current_charges == charge)[0]
            distance = np.linalg.norm(previous_points[previous_indices, None, :] - current_points[None, current_indices, :], axis=2)
            rows, cols = linear_sum_assignment(distance)
            for row, col in zip(rows, cols):
                previous_index = previous_indices[row]
                current_index = current_indices[col]
                current_alive[current_index] = alive[previous_index] and distance[row, col] <= TRACK_MAX_STEP
        previous_points, previous_charges, alive = current_points, current_charges, current_alive
    return float(np.mean(alive)) if len(alive) else 0.0


def run_case(case, seed, dt, detector="vectorized"):
    formation_mode, challenge_mode, edge_i, edge_j, rest, edges = case_spec(case)
    pre = FORMATION[(seed, dt, formation_mode)].copy()
    twin_x = pre.copy(); challenged_x = pre.copy()
    twin_v = np.tile(BULK, (N, 1)); challenged_v = np.tile(BULK, (N, 1))
    challenged_x[GROUP_A] -= DELTA / 2.0; challenged_x[GROUP_B] += DELTA / 2.0
    initial_observed = observe(challenged_x, detector=detector); initial_twin = observe(twin_x, detector=detector)
    initial_error, _ = invariant_geometry_error(initial_observed["points"], initial_observed["charges"], initial_twin["points"], initial_twin["charges"])
    frames = [(initial_observed["points"], initial_observed["charges"])]
    detection_exact = initial_observed["exact"] and initial_twin["exact"]
    max_localization = max(initial_observed["localization"], initial_twin["localization"])
    external_action = 0.0
    steps = int(round(T_TEST / dt)); sample_stride = max(1, int(round(SAMPLE_INTERVAL / dt))); regen_step = int(round(0.5 * T_TEST / dt))
    for step_index in range(steps):
        if challenge_mode == "regen" and step_index == regen_step:
            replacement = twin_x + REGEN_SHIFT
            external_action += float(np.sum(np.linalg.norm(replacement - challenged_x, axis=1)))
            challenged_x = replacement.copy()
        if challenge_mode == "local":
            challenged_force = local_force_vectorized(challenged_x, challenged_v, edge_i, edge_j, rest)
            twin_force = local_force_vectorized(twin_x, twin_v, edge_i, edge_j, rest)
        elif challenge_mode == "global":
            challenged_force = global_force(challenged_x, challenged_v); twin_force = global_force(twin_x, twin_v)
            external_action += dt * float(np.sum(np.linalg.norm(challenged_force, axis=1)))
        else:
            challenged_force = np.zeros_like(challenged_x); twin_force = np.zeros_like(twin_x)
        challenged_v += dt * challenged_force; challenged_x += dt * challenged_v
        twin_v += dt * twin_force; twin_x += dt * twin_v
        if (step_index + 1) % sample_stride == 0:
            current = observe(challenged_x, detector=detector); current_twin = observe(twin_x, detector=detector)
            frames.append((current["points"], current["charges"]))
            detection_exact = detection_exact and current["exact"] and current_twin["exact"]
            max_localization = max(max_localization, current["localization"], current_twin["localization"])
    final_observed = observe(challenged_x, detector=detector); final_twin = observe(twin_x, detector=detector)
    final_error, final_angle = invariant_geometry_error(final_observed["points"], final_observed["charges"], final_twin["points"], final_twin["charges"])
    return {"case": case, "seed": seed, "dt": dt, "initial_error": float(initial_error), "final_error": float(final_error), "R_geometry": float(1.0 - final_error / max(initial_error, 1e-12)), "final_angle_deg": float(final_angle), "track_survival": continuity_score(frames), "external_action": float(external_action), "detection_exact": bool(detection_exact), "max_localization_error": float(max_localization), "challenged_x": challenged_x, "twin_x": twin_x}


def transformed(points, angle=0.0, shift=None, reflect=False):
    matrix = np.array([[math.cos(angle), -math.sin(angle)], [math.sin(angle), math.cos(angle)]], dtype=float)
    result = points.copy()
    if reflect: result[:, 0] *= -1.0
    result = result @ matrix.T
    if shift is not None: result += np.asarray(shift, dtype=float)
    return result


def random_gas(seed):
    rng = np.random.default_rng(seed + 9000)
    for _ in range(10000):
        angle = rng.uniform(0.0, 2.0 * np.pi, N); radius = rng.uniform(1.15, 2.30, N)
        points = np.column_stack((radius * np.cos(angle), radius * np.sin(angle))); points -= points.mean(axis=0)
        pair = np.linalg.norm(points[:, None, :] - points[None, :, :], axis=2) + np.eye(N) * 100.0
        if np.min(pair) > 0.55: return points
    raise RuntimeError("Could not generate separated gas control.")
