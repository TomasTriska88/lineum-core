def decode_unicode15(payload: str, expected_bytes: int) -> bytes:
    acc = bits = 0
    out = bytearray()
    for char in "".join(payload.split()):
        value = ord(char) - BASE
        if not 0 <= value < (1 << 15):
            raise ValueError(f"Invalid Unicode15 code point: U+{ord(char):04X}")
